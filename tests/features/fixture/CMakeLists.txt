cmake_minimum_required(VERSION 3.28)
project(laghu_feature_fixture LANGUAGES NONE)

if(NOT DEFINED LAGHU_SOURCE OR NOT DEFINED SCENARIO)
  message(FATAL_ERROR "Laghu feature fixture requires LAGHU_SOURCE and SCENARIO")
endif()

include("${LAGHU_SOURCE}/cmake/LaghuFeatures.cmake")
laghu_feature_registry_initialize()

if(SCENARIO STREQUAL "positive_http3")
  laghu_resolve_features(effective
    REQUESTED http3
    AVAILABLE core tls http3)
  laghu_materialize_feature_targets(${effective})
  foreach(feature IN ITEMS core tls http3)
    laghu_feature_target_name("${feature}" target)
    if(NOT TARGET "${target}")
      message(FATAL_ERROR "Laghu feature fixture failed: feature=${feature} rule=selected_target_missing")
    endif()
  endforeach()
  if(TARGET laghu_feature_http2)
    message(FATAL_ERROR "Laghu feature fixture failed: feature=http2 rule=disabled_target_created")
  endif()
elseif(SCENARIO STREQUAL "unknown")
  laghu_resolve_features(effective REQUESTED unknown AVAILABLE core)
elseif(SCENARIO STREQUAL "duplicate")
  laghu_resolve_features(effective REQUESTED tls tls AVAILABLE core tls)
elseif(SCENARIO STREQUAL "unavailable")
  laghu_resolve_features(effective REQUESTED tls AVAILABLE core)
elseif(SCENARIO STREQUAL "conflict")
  set_property(GLOBAL PROPERTY LAGHU_FEATURE_CONFLICTS_core http2)
  laghu_resolve_features(effective REQUESTED http2 AVAILABLE core http2)
elseif(SCENARIO STREQUAL "cycle")
  set_property(GLOBAL PROPERTY LAGHU_FEATURE_DEPENDS_tls http3)
  laghu_resolve_features(effective REQUESTED http3 AVAILABLE core tls http3)
else()
  message(FATAL_ERROR "Laghu feature fixture requires a known scenario; scenario=${SCENARIO}")
endif()
