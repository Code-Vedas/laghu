cmake_minimum_required(VERSION 3.28)
project(laghu_dag_fixture LANGUAGES NONE)

if(NOT DEFINED LAGHU_SOURCE OR NOT DEFINED SCENARIO)
  message(FATAL_ERROR "Laghu DAG fixture requires LAGHU_SOURCE and SCENARIO")
endif()

include("${LAGHU_SOURCE}/cmake/LaghuDependencyDag.cmake")

foreach(node IN LISTS LAGHU_SUBSYSTEM_NODES)
  add_library("fixture_${node}" INTERFACE)
  laghu_register_subsystem_target("fixture_${node}" "${node}")
endforeach()

if(SCENARIO STREQUAL "positive_leaf")
  laghu_link_subsystems(fixture_config fixture_core)
  laghu_link_subsystems(fixture_os fixture_core)
elseif(SCENARIO STREQUAL "positive_protocol")
  laghu_link_subsystems(fixture_protocol fixture_core fixture_config fixture_os)
elseif(SCENARIO STREQUAL "positive_top_and_adapter")
  laghu_link_subsystems(fixture_proxy fixture_core fixture_config fixture_os fixture_protocol fixture_tls fixture_cache fixture_observability)
  laghu_link_subsystems(fixture_adapters fixture_core fixture_config fixture_os fixture_protocol fixture_tls fixture_cache fixture_observability fixture_proxy fixture_control fixture_cli)
elseif(SCENARIO STREQUAL "core_outward")
  laghu_link_subsystems(fixture_core fixture_config)
elseif(SCENARIO STREQUAL "os_protocol")
  laghu_link_subsystems(fixture_os fixture_protocol)
elseif(SCENARIO STREQUAL "forbidden_peer")
  laghu_link_subsystems(fixture_cache fixture_tls)
elseif(SCENARIO STREQUAL "cycle")
  laghu_link_subsystems(fixture_config fixture_core)
  laghu_link_subsystems(fixture_core fixture_config)
elseif(SCENARIO STREQUAL "adapter_direction")
  laghu_link_subsystems(fixture_proxy fixture_adapters)
elseif(SCENARIO STREQUAL "generic_utility")
  add_library(fixture_common INTERFACE)
  laghu_register_subsystem_target(fixture_common common)
else()
  message(FATAL_ERROR "Laghu DAG fixture requires a known scenario; scenario=${SCENARIO}")
endif()
