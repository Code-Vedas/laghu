cmake_minimum_required(VERSION 3.28)
project(laghu_dependency_fixture LANGUAGES CXX)

# Dependency fixtures compile ordinary C++23 sources, never C++ modules. Keep
# this local because nested CMake configurations do not inherit the root setting.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

if(NOT DEFINED LAGHU_SOURCE OR NOT DEFINED SCENARIO)
  message(FATAL_ERROR "Laghu dependency fixture requires LAGHU_SOURCE and SCENARIO")
endif()
if(DEFINED LAGHU_EXPECT_CROSSCOMPILING)
  if(LAGHU_EXPECT_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Laghu dependency fixture cross-toolchain inheritance failed: expected=cross actual=native")
  elseif(NOT LAGHU_EXPECT_CROSSCOMPILING AND CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Laghu dependency fixture cross-toolchain inheritance failed: expected=native actual=cross")
  endif()
endif()

include("${LAGHU_SOURCE}/cmake/LaghuDependencies.cmake")
include("${LAGHU_SOURCE}/cmake/LaghuFeatures.cmake")
include("${LAGHU_SOURCE}/cmake/LaghuBuildVariants.cmake")
laghu_dependency_registry_initialize()

if(SCENARIO STREQUAL "positive_symbol" OR SCENARIO STREQUAL "capability_mismatch")
  add_library(yyjson_fixture STATIC yyjson.cpp)
  target_include_directories(yyjson_fixture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/${SCENARIO}/include")
  laghu_add_dependency_symbol_probe(
    ID yyjson
    TARGET laghu_dependency_fixture_probe
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/${SCENARIO}/include"
    LIBRARIES yyjson_fixture)
elseif(SCENARIO STREQUAL "zlib_compat_enabled")
  laghu_validate_zlib_ng_configuration(ZLIB_COMPAT ON)
elseif(SCENARIO STREQUAL "system_openssl")
  laghu_require_system_dependency(openssl)
elseif(SCENARIO STREQUAL "system_dependency")
  if(NOT DEFINED DEPENDENCY)
    message(FATAL_ERROR "Laghu system dependency fixture requires DEPENDENCY")
  endif()
  laghu_require_system_dependency("${DEPENDENCY}")
elseif(SCENARIO STREQUAL "version_too_old")
  laghu_validate_dependency_version(yyjson 0.7.9)
elseif(SCENARIO STREQUAL "unknown_dependency")
  laghu_validate_dependency_version(unknown 1.0.0)
elseif(SCENARIO STREQUAL "positive_http3")
  laghu_validate_http3_dependency_group(
    TLS_PROVIDER OPENSSL
    DEPENDENCIES openssl ngtcp2 nghttp3
    VERSIONS openssl=3.5.0 ngtcp2=1.25.0 nghttp3=1.18.0)
elseif(SCENARIO STREQUAL "http3_incomplete")
  laghu_validate_http3_dependency_group(
    TLS_PROVIDER OPENSSL
    DEPENDENCIES openssl ngtcp2
    VERSIONS openssl=3.5.0 ngtcp2=1.25.0)
elseif(SCENARIO STREQUAL "http3_mixed_provider")
  laghu_validate_http3_dependency_group(
    TLS_PROVIDER OPENSSL
    DEPENDENCIES openssl libressl ngtcp2 nghttp3
    VERSIONS openssl=3.5.0 libressl=3.9.2 ngtcp2=1.25.0 nghttp3=1.18.0)
elseif(SCENARIO STREQUAL "http3_tls_too_old")
  laghu_validate_http3_dependency_group(
    TLS_PROVIDER OPENSSL
    DEPENDENCIES openssl ngtcp2 nghttp3
    VERSIONS openssl=3.4.9 ngtcp2=1.25.0 nghttp3=1.18.0)
elseif(SCENARIO STREQUAL "mode_defaults" OR SCENARIO STREQUAL "mode_minimal_no_fetch")
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  if(NOT LAGHU_DEPENDENCY_SOURCE STREQUAL VENDORED OR
      NOT LAGHU_DEPENDENCY_LINK_MODE STREQUAL STATIC OR
      NOT LAGHU_TLS_PROVIDER STREQUAL OPENSSL)
    message(FATAL_ERROR "Laghu dependency fixture default modes are not canonical")
  endif()
  if(NOT LAGHU_DEPENDENCY_SELECTION_METADATA)
    message(FATAL_ERROR "Laghu dependency fixture selection metadata is missing")
  endif()
  if(SCENARIO STREQUAL "mode_minimal_no_fetch" AND
      (EXISTS "${CMAKE_BINARY_DIR}/_deps" OR TARGET laghu_dependency_openssl))
    message(FATAL_ERROR "Laghu dependency fixture minimal configuration acquired a dependency")
  endif()
elseif(SCENARIO STREQUAL "mode_minimal_ignored_selectors")
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  file(READ "${LAGHU_DEPENDENCY_SELECTION_METADATA}" selection_metadata)
  foreach(selector IN ITEMS source link_mode tls_provider)
    string(FIND "${selection_metadata}" "\"${selector}\":" selector_offset)
    if(NOT selector_offset EQUAL -1)
      message(FATAL_ERROR
        "Laghu dependency fixture inactive selector was recorded: selector=${selector}")
    endif()
  endforeach()
  if(EXISTS "${CMAKE_BINARY_DIR}/_deps" OR TARGET laghu_dependency_openssl OR
      TARGET laghu_dependency_libressl)
    message(FATAL_ERROR "Laghu dependency fixture ignored selectors acquired a dependency")
  endif()
elseif(SCENARIO STREQUAL "mode_full_selection")
  set(LAGHU_BUILD_PROFILE FULL)
  set(LAGHU_FEATURES "")
  laghu_configure_build_variant()
  laghu_select_active_dependencies(active
    FEATURES ${LAGHU_EFFECTIVE_FEATURES}
    TLS_PROVIDER "${LAGHU_TLS_PROVIDER}")
  foreach(dependency IN ITEMS nghttp2 ngtcp2 nghttp3 yyjson c_ares pcre2_8bit zlib_ng brotli zstd libmaxminddb)
    list(FIND active "${dependency}" dependency_index)
    if(dependency_index EQUAL -1)
      message(FATAL_ERROR
        "Laghu dependency fixture full profile dependency missing: dependency=${dependency}")
    endif()
  endforeach()
  string(TOLOWER "${LAGHU_TLS_PROVIDER}" selected_tls)
  if(selected_tls STREQUAL openssl)
    set(excluded_tls libressl)
  else()
    set(excluded_tls openssl)
  endif()
  list(FIND active "${selected_tls}" selected_tls_index)
  list(FIND active "${excluded_tls}" excluded_tls_index)
  if(selected_tls_index EQUAL -1 OR NOT excluded_tls_index EQUAL -1)
    message(FATAL_ERROR "Laghu dependency fixture full profile TLS provider is not exclusive")
  endif()
elseif(SCENARIO STREQUAL "mode_tls_openssl" OR SCENARIO STREQUAL "mode_tls_libressl")
  if(SCENARIO STREQUAL "mode_tls_openssl")
    set(provider OPENSSL)
    set(expected openssl)
    set(excluded libressl)
  else()
    set(provider LIBRESSL)
    set(expected libressl)
    set(excluded openssl)
  endif()
  laghu_select_active_dependencies(active FEATURES tls TLS_PROVIDER "${provider}")
  list(FIND active "${expected}" expected_index)
  list(FIND active "${excluded}" excluded_index)
  if(expected_index EQUAL -1 OR NOT excluded_index EQUAL -1)
    message(FATAL_ERROR "Laghu dependency fixture TLS provider selection is not exclusive")
  endif()
elseif(SCENARIO STREQUAL "mode_static_artifact_nonstatic")
  set(nonstatic "${CMAKE_BINARY_DIR}/libfixture.dylib")
  file(WRITE "${nonstatic}" "not an archive")
  laghu_prove_static_artifact(yyjson "${nonstatic}")
elseif(SCENARIO STREQUAL "mode_invalid_source" OR
       SCENARIO STREQUAL "mode_invalid_link_mode" OR
       SCENARIO STREQUAL "mode_invalid_tls_provider")
  laghu_validate_dependency_modes()
elseif(SCENARIO STREQUAL "mode_system_dynamic")
  set(LAGHU_BUILD_PROFILE CUSTOM)
  set(LAGHU_FEATURES tls)
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  if(NOT TARGET laghu_dependency_openssl)
    message(FATAL_ERROR "Laghu dependency fixture system TLS target is missing")
  endif()
elseif(SCENARIO STREQUAL "mode_vendored_tls")
  set(LAGHU_BUILD_PROFILE CUSTOM)
  set(LAGHU_FEATURES tls)
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  if(NOT TARGET laghu_dependency_openssl OR TARGET laghu_dependency_libressl)
    message(FATAL_ERROR "Laghu dependency fixture vendored TLS provider is not exclusive")
  endif()
elseif(SCENARIO STREQUAL "mode_vendored_yyjson")
  set(LAGHU_BUILD_PROFILE CUSTOM)
  set(LAGHU_FEATURES structured_data)
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  if(NOT TARGET laghu_dependency_yyjson)
    message(FATAL_ERROR "Laghu dependency fixture vendored yyjson target is missing")
  endif()
  add_executable(laghu_dependency_mode_consumer consumer.cpp)
  target_link_libraries(laghu_dependency_mode_consumer PRIVATE laghu_dependency_yyjson)
elseif(SCENARIO STREQUAL "mode_vendored_http3_transport")
  # ngtcp2 declares this option ON by default.  Preserve that upstream default
  # in the fixture to prove Laghu isolates its transport build from host TLS.
  set(ENABLE_OPENSSL ON CACHE BOOL "ngtcp2 crypto backend default" FORCE)
  set(LAGHU_BUILD_PROFILE CUSTOM)
  set(LAGHU_FEATURES http3)
  laghu_configure_build_variant()
  laghu_configure_dependency_modes()
  if(NOT TARGET laghu_dependency_ngtcp2 OR NOT TARGET laghu_dependency_nghttp3)
    message(FATAL_ERROR "Laghu dependency fixture vendored HTTP/3 transport targets are missing")
  endif()
else()
  message(FATAL_ERROR "Laghu dependency fixture requires a known scenario; scenario=${SCENARIO}")
endif()
